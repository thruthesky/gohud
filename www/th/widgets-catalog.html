วิดเจ็ตทั้งหมด

วิดเจ็ตและเลย์เอาต์ทั้งหมดของ gohud ตามชื่อ พร้อมตัวอย่างที่วางใช้ได้ทันที ทุกตัวใช้รูปลักษณ์ของพรีเซ็ตที่เลือกอยู่ ← วิดเจ็ต

วิดเจ็ต วิดเจ็ตทั้งหมด หน้าต่างลอย การบอกอะไรบางอย่างกับผู้เล่น HUD การรอ การบอก การนับ ฟอร์มและรายการ รูปทรงที่เกมใช้จริง ทับอยู่บนหน้าจอ GoStyle — ที่เดียวที่ทุกอย่างถูกสร้างขึ้น

เลย์เอาต์ (15)

GoScaffold

หน้าจอแอปที่ต่อไว้ครบ — แถบบน หน้าเลื่อน แถบล่าง FAB และลิ้นชัก

สืบทอดจาก Control

var page := GoStyle.column()
var screen := GoScaffold.make("Inbox", page, true)
screen.set_fab(GoFab.make(GoIconSet.EDIT))
add_child(screen)

GoTopBar

รายการตามขอบบนในหนึ่งถึงสามช่อง ช่องกลางอยู่ตรงกลางเสมอ

สืบทอดจาก GoEdgeBar

var bar := GoTopBar.make(3)
bar.add_start(GoStyle.icon_button(GoIconSet.BACK, go_back, -1, &"back"))
bar.add_center(GoStyle.label("Stage 3"))
bar.add_end(GoStyle.chip("1,250"))

GoBottomBar

รายการตามขอบล่าง หลบแถบท่าทาง

สืบทอดจาก GoEdgeBar

var actions := GoBottomBar.make(1, GoBottomBar.Justify.SPACE_BETWEEN)
for name in ["Attack", "Guard", "Run"]: actions.add_start(GoStyle.button(name))
screen.add_child(actions)

GoLeftSideBar

สูงสุดสามชั้นตามขอบซ้าย

สืบทอดจาก GoSideBar

var tools := GoLeftSideBar.make(3)
tools.add_start(GoStyle.icon_button(GoIconSet.MENU, open_menu, -1, &"menu"))
tools.add_end(GoStyle.icon_button(GoIconSet.USER, open_profile, -1, &"profile"))
screen.add_child(tools)

GoRightSideBar

สูงสุดสามชั้นตามขอบขวา

สืบทอดจาก GoSideBar

var tools := GoRightSideBar.make(3)
tools.add_start(GoStyle.icon_button(GoIconSet.SETTINGS, open_settings, -1, &"settings"))
tools.add_center(GoStyle.icon_button(GoIconSet.SEARCH, find, -1, &"search"))
screen.add_child(tools)

GoSideBar

แถบข้างที่เป็นฐานของทั้งสอง คงด้านเดิมแม้ในภาษาที่เขียนจากขวาไปซ้าย

สืบทอดจาก GoEdgeBar

var rail := GoRightSideBar.make(1, GoSideBar.Justify.CENTER)
rail.add_start(GoStyle.icon_button(GoIconSet.MAP, open_map, -1, &"map"))

GoEdgeBar

กติกาที่แถบขอบทุกแบบใช้ร่วมกัน: ช่อง การกระจาย การยึดขอบ และพื้นที่ปลอดภัย

สืบทอดจาก Container

var bar := GoTopBar.make(1)
bar.justify = GoEdgeBar.Justify.SPACE_BETWEEN
bar.pin_to_edge = GoEdgeBar.Pin.ALWAYS

GoGrid

คอลัมน์กว้างเท่ากันพอดี — จำนวนตายตัว หรือเท่าที่ใส่ได้

สืบทอดจาก Container

var cards := GoGrid.make(1, 160.0)
for item in items: cards.add_child(GoStyle.item_card(item))

GoForm

ฟอร์มที่จำกัดความกว้างและหลบคีย์บอร์ดกับ HUD

สืบทอดจาก MarginContainer

var form := GoForm.new()
form.avoid_hud = true

add_child(form)

GoScroll

เลื่อนด้วยการสัมผัสได้จากทุกจุดของเนื้อหา รองรับภาษาที่เขียนจากขวาไปซ้าย

สืบทอดจาก ScrollContainer

var strip := GoScroll.horizontal()

ตรึงชิ้นส่วน HUD ไว้ที่หนึ่งในเก้าตำแหน่งของพื้นที่ปลอดภัย

สืบทอดจาก Control

corner.spot = GoHudAnchor.Spot.TOP_LEFT
corner.add_child(GoStyle.hud_panel())

GoStyle.column · row · wrap_row

คอลัมน์ แถว และแถวที่ขึ้นบรรทัดใหม่ได้ ด้วยระยะห่างจากธีม

กริดที่เพิ่มคอลัมน์เมื่อกว้างขึ้น และกล่องที่รักษาสัดส่วน

GoStyle.padding · spacer · divider

ระยะขอบใน ช่องว่างที่กินพื้นที่ที่เหลือ และเส้นคั่น

สืบทอดจาก GoStyle

GoStyle.foldable · section

สืบทอดจาก GoStyle
