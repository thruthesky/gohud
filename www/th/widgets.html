วิดเจ็ต

ทุกตัวสร้างด้วย .new() แล้ว add_child() — ไม่มีไฟล์ซีน ไม่มี autoload หมายเหตุที่มี 🛑 กำกับคือกับดักที่เจอมาแล้วบนหน้าจอจริง ไม่ใช่ทฤษฎี

หน้าต่างลอย

GoSurface — เปลือกนอก

ป๊อปอัป ชีต และดรอปดาวน์ ล้วนใช้ตัวนี้ตัวเดียว มันจัดการเซฟแอเรีย คีย์บอร์ดเสมือน หัวและท้ายที่ติดหนึบ เนื้อหาที่เลื่อนได้ การตัดสินว่าหน้าต่างไหนอยู่บนสุด การคืนโฟกัส ความเป็นเจ้าของปุ่มย้อนกลับ และการลากปรับขนาด

ตำแหน่งหน้าตาเป็นอย่างไรใช้กับอะไร
CENTERการ์ดกลางจอกล่องยืนยัน หน้าตั้งค่า
BOTTOMชีตที่ยกขึ้นมาจากด้านล่างรายการต่าง ๆ หน้าจัดการ
ANCHORการ์ดที่ตรึงไว้ข้างคอนโทรลหนึ่งดรอปดาวน์ เมนูคลิกขวา
มันไม่ลบตัวเองทิ้ง การตอบสนองต่อ close_requested — จะซ่อนหรือจะคืนหน่วยความจำ — เป็นหน้าที่ของหน้าจอที่เป็นเจ้าของ เพราะตัวเปลือกไม่มีทางรู้ว่ามันกำลังถูกปิดด้วยเหตุผลอะไร (บันทึกแล้วปิด หรือทิ้งไปเลย)

GoSheet — หน้าที่มาจากด้านล่าง

var sheet := GoSheet.new()
add_child(sheet)
sheet.open("กระเป๋าของ")
sheet.toolbar().add_child(GoStyle.line_edit("ค้นหา…"))
sheet.toolbar().visible = true
sheet.footer().add_child(GoStyle.button("ปิด", sheet.close, GoStyle.Tone.PRIMARY))
sheet.footer().visible = true

มันเป็นเจ้าของ CanvasLayer ของตัวเอง จึงลอยอยู่เหนือ HUD ได้อย่างแน่นอน open() จะล้างเนื้อหาในตัวชีตและปิดปุ่มย้อนกลับกับแถวติดหนึบ — ปุ่มที่กดไม่ได้บนหน้าจอที่ไม่มีที่ให้ย้อนกลับ ทำให้คนอ่านว่าแอปพัง

GoDialogs — กล่องยืนยันที่ await ได้

if await dialogs.confirm("ลบไฟล์เซฟ", "การกระทำนี้ย้อนกลับไม่ได้"):
    delete_save()

confirm(title, body, ok_text, cancel_text, extra, args, destructive) — ทุกอย่างหลังจากเนื้อความ เป็นตัวเลือกทั้งหมด และ alert() ทำงานแบบเดียวกันแต่มีปุ่มเดียว ตั้ง destructive สำหรับสิ่งที่ย้อนกลับไม่ได้: ปุ่มยืนยันจะกลายเป็นปุ่มทึบ GoStyle.Tone.DANGER_SOLID บนธีมสว่าง ป้ายข้อความสีอันตรายแบบย้อมสีจำเป็นต้องเข้มลงจนอ่านเหมือนตัวอักษรสีดำ แต่ปุ่มแบบทึบ ยังคงตัวอักษรสีขาวบนพื้นแดงไว้ได้ โทนสีถูกกำหนดใหม่ทุกครั้งที่เรียกเพราะหน้าต่างถูกใช้ซ้ำอยู่บานเดียว และการเรียก confirm() ครั้งที่สองขณะที่ยังมีบานเปิดค้างอยู่จะคืนค่า false ทันที ส่วน confirm_key() รับคีย์คำแปลเข้าไปแทน

การจัดวางปุ่ม action_layout มีค่าเริ่มต้นเป็น VERTICAL (ปลอดภัยสำหรับคำแปลยาว ๆ) ใช้ HORIZONTAL เพื่อจัดเป็นแถวเดียวโดยแบ่งความกว้างคนละครึ่ง หรือ AUTO ซึ่งจะใช้แถวเดียวก็ต่อเมื่อป้ายข้อความทั้งสองพอดีกับครึ่งหนึ่งของการ์ดในบรรทัดเดียว ไม่อย่างนั้นก็ซ้อนกันลงมา action_gap คือช่องไฟระหว่างปุ่ม ส่วน body_gap คือช่องไฟเหนือปุ่ม — ตั้งให้กว้างกว่า ช่องไฟระหว่างปุ่ม เพื่อให้คำถามกับตัวเลือกอ่านออกมาเป็นสองส่วน ถ้าอยากเปลี่ยนเฉพาะไดอะล็อกบานเดียว ให้เรียก set_next_action_layout() ก่อนเปิดมัน แถวปุ่มถูกสร้างตอนที่ไดอะล็อกเปิดครั้งแรก การใส่ GoDialogs เป็น autoload จึงไม่เพิ่มภาระตอนเปิดเกมเลย

GoForm — ฟอร์มที่จำกัดความกว้าง

มันรับประกันว่าป้ายข้อความที่เป็นลูกหลานของมันจะตัดบรรทัด ถ้าไม่มีสิ่งนี้ ประโยคยาว ๆ ประโยคเดียว จะยืดเป็นบรรทัดเดียวที่ความกว้างขั้นต่ำเกินหน้าจอ แล้วสองข้างจะถูกตัดหาย — จนดูไม่ออกด้วยซ้ำว่า คุณอยู่หน้าจอไหน ส่วนการไล่เปิดการตัดบรรทัดด้วยมือทีละซีนนั้น ลืมทุกครั้ง

GoScroll — การเลื่อนด้วยนิ้ว

ใส่ของไว้ข้างในแค่ ชิ้นเดียว ส่วนหัวและแถวปุ่มต้องอยู่ข้างนอก — รายการยาว ๆ ต้องไม่ดันมันตกจอไป use_panel_edge() ย้ายแถบเลื่อนเข้าไปอยู่ในระยะขอบที่การ์ดมีอยู่แล้ว ขณะที่เนื้อหายังคงระยะย่อหน้าเดิมไว้

การบอกอะไรบางอย่างกับผู้เล่น

GoNotice — สแน็กบาร์

มันไม่ดักอินพุตเด็ดขาด ข้อความแจ้งเตือนมีไว้อ่าน ไม่ได้มีไว้กด มันไม่แย่งโฟกัส ไม่ขวางอินพุตของเกม และปุ่มที่อยู่ข้างใต้ก็ยังกดได้ ถ้าต้องการการยืนยันให้ใช้ GoDialogs ถ้าต้องการให้เลือกให้ใช้ GoPromptCard

GoPromptCard — คำถามที่ไม่หยุดเกม

ลอยอยู่ด้านข้างโดยไม่มีฉากหลังทึบ เกมยังเดินต่อไปตามปกติ

GoCoachMark — ทัวร์แนะนำ

ชี้ไปยังคอนโทรลจริงทีละอัน มีเพียงการ์ดเท่านั้นที่รับอินพุต ดังนั้น การกดคอนโทรลที่ถูกไฮไลต์เอง จะพาทัวร์เดินหน้าต่อ — "กดตรงนี้" จึงกลายเป็นการลงมือทำ แทนที่จะเป็นแค่ประโยคหนึ่ง

tour.start([
    {"target": bag_button, "title": "กระเป๋า", "body": "ของทุกอย่างที่เก็บได้จะมาอยู่ที่นี่"},
    {"target": map_button, "title": "แผนที่", "body": "กดเพื่อเปิดแผนที่เต็ม"},
])

title และ body จะเป็นคีย์คำแปลหรือข้อความธรรมดาก็ได้ ขั้นตอนที่เป้าหมาย หายไปแล้วจะถูกข้ามไป

HUD

เรื่องเล็ก ๆ ต่างหากที่ตัดสินว่าหน้าจอดูดีหรือไม่

จะขัดเกลาพาเนลใหญ่ ๆ แค่ไหนก็ตาม สิ่งที่สะดุดตาเสมอคือเรื่องเล็ก ๆ: คำที่ถูกหักเป็นสองท่อน แสงเรืองที่ถูกตัดขาด สล็อตที่แน่นจนอึดอัด มีสี่เรื่องแบบนี้ถูกชี้ให้เห็นเมื่อ 2026-09-13 และถูกแก้ที่ตัววิดเจ็ตกับธีม ไม่ใช่แก้ที่หน้าจอเดโม ทุกหน้าจอจึงได้ผลลัพธ์เดียวกัน

อาการที่เห็นสาเหตุวิธีแก้
ปุ่ม Done ของโค้ชมาร์กกลายเป็น Don/e ปุ่มที่เปิดการตัดบรรทัดจะหักความกว้างของข้อความออกจากความกว้างขั้นต่ำของมัน พอถูกบีบ มันจึงหักกลางคำ GoStyle.fit_words(): คำเดียวไม่มีวันถูกตัดบรรทัด ถ้ามีหลายคำก็ตัดบรรทัดได้ แต่คำที่ยาวที่สุดต้องพอดีเสมอ โดยตัดสินจากข้อความที่แสดงจริง ไม่ใช่จากคีย์คำแปล และฟอร์มจะคำนวณใหม่ให้เมื่อเปลี่ยนภาษา
แสงเรืองของปุ่มเต็มความกว้างถูกตัดหายไปในแนวตั้งทางด้านซ้าย คอนเทนเนอร์ที่เลื่อนได้จะตัดทุกอย่างที่ขอบของมันโดยไม่มีข้อยกเว้น เหลือรอดแค่ด้านขวาเพราะมีร่องของรางเลื่อนช่วยไว้ GoScroll ขอยืมระยะขอบของพาเรนต์มาดันขอบของมันออกไป 12dp แล้วคืนกลับเข้ามาข้างใน — เนื้อหาอยู่ที่เดิม
เวลา ไอคอน และจำนวน ซ้อนกันเป็นแนวตั้งในสล็อต สามแถวอัดอยู่ใน 48dp ไอคอนขนาดใหญ่อยู่กลาง ปุ่มลัดอยู่มุมบนซ้าย จำนวนเป็นแบดจ์มุมล่างขวา และเวลาที่เหลือเป็นแบดจ์ทับบนไอคอน — พาเนลของแบดจ์มาจาก badge_box() ของสกิน
ดรอปดาวน์ที่แบนราบจนมองไม่ออกว่าเลือกอะไรอยู่ การชี้ที่รายการใช้พาเนลของปุ่ม ส่วนเครื่องหมายเรดิโอเป็นของเอนจินที่จางมาก พาเนลชี้เฉพาะของมันเอง (ย้อมสีเน้นอ่อน ๆ) เครื่องหมายเรดิโอชุดเดียวกับเช็กบ็อกซ์ และความสูงของรายการที่พอดีกับการกดด้วยนิ้ว

ใส่คำอธิบายให้ปุ่มที่มีแต่ไอคอนด้วย GoStyle.icon_button(icon, action, -1, tooltip_key) — ทั้งทูลทิปสำหรับคนใช้เมาส์และชื่อเชิงการเข้าถึงสำหรับโปรแกรมอ่านหน้าจอ มาจากค่าเดียวกันนี้ gohud วาดทูลทิปเอง เพราะของเอนจินจะฉีกข้อความเป็นบรรทัดละตัวอักษรเมื่อการคำนวณความกว้าง ของมันผิดพลาด (วัดได้ว่ากว้างแค่ 1dp)

การ์ดโค้ชมาร์กหลบสิ่งที่ตรึงอยู่กับที่

เมื่อวางแนวนอน การ์ดเคยถูกส่งไปที่ขอบบนสุดหรือล่างสุด — ซึ่งเป็นที่อยู่ของแถบหัวและชิ้นส่วน HUD พอดี ในเดโมมันไปนั่งทับ → × ของแถบหัว ตอนนี้มันอยู่ระดับเดียวกับเป้าหมาย เดินอ้อม GoHudAnchor ทุกอันที่จองพื้นที่ไว้ และรับสิ่งอื่นที่ต้องหลบผ่าน keep_clear มันขยับด้วยระยะที่สั้นที่สุดที่พ้นสิ่งกีดขวางโดยไม่บังเป้าหมาย และจะอยู่นิ่งถ้าหน้าจอเล็กเกินกว่าจะหลบได้ครบ — ทับกันยังดีกว่าหายไปเลย ความลึกของมัน — ที่ทำให้มันอ่านว่าลอยอยู่เหนือหน้าเนื้อหา — มาจากค่าปรับของสกิน (float_shadow_size, float_shadow_lift, float_shadow_alpha; สกินมุมตัดเฉียงใช้ float_glow_size) 

tour.keep_clear = [header_bar, toolbar]   # สิ่งที่ตรึงอยู่กับที่แต่ไม่ใช่แองเคอร์

หน้าจอที่มีเมาส์และคีย์บอร์ด

นิ้วไม่มีสถานะ hover ถ้าพัฒนาโดยคิดถึงแต่มือถือ จะมีสองสถานะที่มีเฉพาะบนเดสก์ท็อปหลุดการตรวจไป — และหนึ่งในนั้นก็พังจริง ๆ: วงแหวนโฟกัสของ gohud มองไม่เห็นเลยบนปุ่มหลัก เพราะพาเนลใช้สีเน้น และวงแหวนก็ใช้สีเน้นเหมือนกัน: 1.00:1 คือสีเดียวกันวาดทับตัวเอง เมื่อไล่ด้วยคีย์บอร์ด จึงไม่มีทางรู้เลยว่าตอนนี้อยู่ตรงไหน

ปุ่มแบบทึบ (GoPrimaryButton, GoDangerSolidButton) ใช้วงแหวนที่ ตัดกับพาเนลของตัวเอง — สีเข้มบนธีมมืด สีขาวบนธีมสว่าง และวงเล็บเล็งเป้าสีเข้มในธีมไซไฟ ตัวตรวจคอนทราสต์วัดวงแหวนโฟกัสเทียบกับพาเนลที่มันวางอยู่ กฎนี้จึงอยู่รอดแม้เปลี่ยนจานสี (WCAG 2.4.11 กำหนดไว้ที่ 3:1 สำหรับตัวบอกโฟกัส)

วิดเจ็ต HUD

GoHudAnchor — เก้าตำแหน่ง

เคารพเซฟแอเรียและระยะขอบจอ ใส่ landscape_spot ให้มันแล้วมันจะย้ายที่เมื่อวางแนวนอน — ปุ่มควบคุมที่อยู่กลางล่างในแนวตั้งจึงไปอยู่มุมล่างขวาแทนได้

ทำไมต้องมีมัน ถ้าเขียนแองเคอร์และออฟเซ็ตด้วยมือทีละชิ้น HUD ค่าจะเพี้ยนไปเรื่อย ๆ: แถบพลังชีวิตไปโผล่ใต้รอยบากกล้องบนเครื่องที่มีนอตช์ และปุ่มควบคุมไปตกบนแถบท่าทางเมื่อวางแนวนอน เลขคณิตพวกนั้นถูกรวมไว้ที่เดียว

กันเนื้อหาไม่ให้มุดเข้าไปใต้ HUD

เมื่อหน้าจอที่เลื่อนได้ต้องแบ่งพื้นที่กับ HUD เนื้อหาจะเลื่อนลอดเข้าไปข้างใต้ แล้วข้อความสองชุด จะทับกันจนอ่านไม่ออกทั้งคู่ เรื่องนี้เกิดขึ้นสองครั้งในแอดออนนี้: ข้อความตัวอย่างในช่องกรอกถูกบีบ โผล่ออกมาระหว่างควิกสล็อต และปุ่มสวิตช์ในหน้าตั้งค่าไปตกบนแถบพลังชีวิต 

form.avoid_hud = true                  # หลบทุก GoHudAnchor ที่มองเห็นอยู่
joystick_anchor.reserve_space = false  # …ยกเว้นอันที่โผล่มาเฉพาะตอนมีนิ้วแตะ

สี่เหลี่ยมแต่ละผืนจะถูกหลบไปในทิศที่เสียพื้นที่น้อยที่สุด แถบที่อยู่มุมบนขวาจะถูกหลบลงล่าง บนมือถือแนวตั้ง (คิดเป็น 13% ของความสูง) และหลบไปด้านข้างเมื่อวางแนวนอน (21% ของความกว้าง) — เพราะถ้าหลบในแนวตั้งตอนวางแนวนอนจะเสียพื้นที่ถึง 27% ของหน้าจอ พื้นที่ที่ยอมเสียไปแล้ว จะถูกหักออกก่อนวัดสี่เหลี่ยมผืนถัดไป

ชิ้นส่วน HUD ก็ชนกันเองได้เหมือนกัน

เก้าตำแหน่งนั้นแบ่งหน้าจอออกเป็นส่วน ๆ แต่ไม่ได้รับประกันว่าชิ้นส่วนแต่ละชิ้นจะไม่ชนกัน สแน็กบาร์กว้าง ๆ ที่กลางบนเคยไปลงเต็ม ๆ บนแถบพลังชีวิตที่มุมบนขวา จนบังตัวเลขไว้หมด — อ่านไม่ออกเลยว่าเหลือพลังชีวิตเท่าไร 

notice_anchor.avoid_peers = true      # ไปลงใต้ HUD ที่อยู่กับที่
notice_anchor.reserve_space = false   # และไม่ดันหน้าเนื้อหาระหว่างที่มันโผล่อยู่

มันหลบในแนวตั้งเท่านั้น — ถ้าหลบด้านข้างด้วย toast ที่จัดกึ่งกลางจะไปโผล่คนละที่ทุกครั้งที่แสดง และมันหลบเฉพาะชิ้นส่วนที่ไม่ได้กำลังหลบอยู่เหมือนกัน: ถ้าต่างฝ่ายต่างยอมหลบ ทั้งคู่จะสลับที่กันไปไม่มีวันจบ

GoBar — พลังชีวิต มานา ค่าประสบการณ์

var hp := GoBar.new()
hp.label_text = "HP"
hp.ink = GoUi.color(GoTheme.DANGER_FILL)   # โทเคนที่ใช้เป็นสีเติมอย่างเดียว
hp.set_values(320, 500)                     # "320 / 500"

การเปลี่ยนค่าจะไหลนุ่มนวลใน 0.18 วินาที — แถบพลังชีวิตที่กระโดดพรวดไม่ได้บอกคุณว่าโดนหนักแค่ไหน ตั้ง reduce_motion แล้วมันจะเปลี่ยนทันที ตัวเลขใหญ่ ๆ จะถูกย่อเป็น 12.3k เพื่อไม่ให้ความกว้างของแถบกระตุกไปมาตามจำนวนหลักที่เพิ่มลด

GoSlot — ควิกสล็อตหนึ่งช่อง

ไอคอน จำนวน คูลดาวน์ และปุ่มลัด อยู่บนหน้าเดียวกัน การเอาแบดจ์ไปห้อยรอบวงกลมเล็ก ๆ จะทำให้แต่ละสล็อตกว้างไม่เท่ากัน แล้วทั้งแถวก็โยกเยก

ขนาดที่เห็นกับขนาดที่กดได้ไม่เท่ากัน สล็อตที่เรียงชิดกันมีกล่องสัมผัส 48dp ที่ซ้อนทับกัน พื้นที่ซ้อนนั้นจะตกเป็นของสล็อตที่จุดกึ่งกลางใกล้กว่า — บอกให้พวกมันรู้จักกันผ่าน touch_peers

keyboard_focus ปิดไว้เป็นค่าเริ่มต้น สล็อตแปดช่องเรียงกันจึงไม่ไปขวางระหว่างผู้ใช้คีย์บอร์ด กับคอนโทรลในหน้าตั้งค่าทุกตัวในลำดับ Tab เปิดมันเมื่อคีย์บอร์ดหรือเกมแพดเป็นอินพุตเดียวที่มี

GoJoystick — จอยสติ๊กเสมือน

โหมดพฤติกรรม
FIXEDอยู่ที่เดิมเสมอ — หาเจอง่ายโดยไม่ต้องมอง
FOLLOWโผล่ขึ้นตรงที่นิ้วโป้งลง
RELATIVEตามนิ้วต่อไปเรื่อย ๆ หลังโผล่ขึ้นมาแล้ว — สบายสำหรับการเคลื่อนที่นาน ๆ

Vector2 ที่ได้จาก moved มีความยาว 0–1 คุณจึงคูณเข้ากับความเร็วได้ตรง ๆ และค่านี้ไม่แปรตามขนาดหน้าจอหรือ DPI

GoIconButton — ดูเล็ก แต่กดง่าย

ปลอดภัยเฉพาะตอนที่เพื่อนบ้านของมันไม่รับอินพุต ถ้าพื้นที่ที่ขยายออกไปครอบปุ่มถัดไป ปุ่มนั้นจะหยุดตอบสนอง ปุ่มไอคอนที่วางเรียงชิดกันต้องใช้ touch_peers

การรอ การบอก การนับ

GoSnackbar — ข้อความที่จัดที่ให้ตัวเอง

วางตัวอยู่ด้านล่าง (หรือด้านบน) เหนือเซฟแอเรียและคีย์บอร์ด แล้วหายไปเอง ถ้าไม่มีมัน คุณต้องเขียนเลเยอร์ การจัดตำแหน่ง ตัวจับเวลา และการเก็บกวาด ซ้ำในทุกหน้าจอที่ต้องบอกว่า "บันทึกแล้ว"

snack.show_text("บันทึกแล้ว", GoTheme.SUCCESS)

# โอกาสที่จะย้อนกลับ — 0 คือปุ่มแรก, -1 หมายถึงหมดเวลาไปเอง
if await snack.post({"text": "ทิ้งไอเทมแล้ว", "actions": ["ย้อนกลับ"]}) == 0:
    restore_item()

ข้อความจะต่อคิวแทนที่จะเขียนทับกัน และบรรทัดเดิมที่ซ้ำกันจะถูกรวมเป็นอันเดียว — เซิร์ฟเวอร์ที่ ล้มเหลววินาทีละสี่ครั้งจะไม่กองการแจ้งเตือนทิ้งไว้สี่นาทีอีกต่อไป สแน็กบาร์ที่ไม่มีปุ่มจะปล่อยอินพุตผ่าน เกมที่อยู่ข้างใต้จึงเดินต่อไปตามปกติ

มันไม่ใช่ที่สำหรับถามอะไร มันหายไปเอง ผู้เล่นจึงอาจไม่ทันเห็นมันเลย การยืนยันที่ย้อนกลับไม่ได้ เป็นหน้าที่ของ GoDialogs ปุ่มที่นี่ต้องเป็นตัวเลือกเสริมเท่านั้น — ย้อนกลับ ดูรายละเอียด ลองใหม่

GoNotice คืออีกครึ่งหนึ่งของคู่นี้: มันไม่รับอินพุตหรือโฟกัสเด็ดขาด และหน้าจอที่เป็นเจ้าของมัน เป็นคนตัดสินว่ามันจะไปอยู่ตรงไหน มันใส่ปุ่มไม่ได้ แต่ตัวนี้ใส่ได้

GoSpinner — การรอที่ยังมองไม่เห็นปลายทาง

เกมที่คุยกับเซิร์ฟเวอร์ต้องการตัวนี้มากกว่าอะไรทั้งหมดในหน้านี้: หน้าจอที่นิ่งอยู่เฉย ๆ คนอ่านว่าเกมค้าง เรียกครั้งเดียวก็เปลี่ยนปุ่มที่คุณเพิ่งกดให้เป็นสปินเนอร์ตรงที่เดิม

GoSpinner.busy(buy_button, true)
var ok := await server.purchase(item)
GoSpinner.busy(buy_button, false)

ปุ่มยังคงขนาดเดิม — ป้ายข้อความกลายเป็นโปร่งใสแทนที่จะหายไป แถวจึงไม่กระโดด — และมันถูกปิดการกด ขณะที่กำลังหมุน ส่วนหลังนั่นแหละคือประเด็น: การซื้อต้องไม่ถูกยิงสองครั้งระหว่างที่คุณ await อยู่ ถ้าเปิด reduce_motion มันจะไม่หมุน แต่เป็นจุดสามจุดเต้นเป็นจังหวะแทน

ความคืบหน้าที่วัดได้คือ GoBar ส่วนสปินเนอร์ไม่ได้สัญญาอะไรเลยว่ามันจะจบเมื่อไร

GoBadge — จุดที่ยังไม่ได้อ่าน ป้าย "ใหม่" และเลข 99+

GoBadge.attach(mail_button, unread_count)   # ซ่อนตัวเองเมื่อเป็นศูนย์
GoBadge.attach(shop_button, 0, "ใหม่")
GoBadge.attach(friend_button, 3, "", true) # จุดจุดเดียว — "มีอะไรอยู่ตรงนั้น"

มันห้อยอยู่ที่มุมบนขวาของโฮสต์ด้วยแองเคอร์ จึงตามโฮสต์ที่ขนาดถูกกำหนดทีหลังได้ และการแนบซ้ำครั้งที่สอง จะคืนแบดจ์อันเดิม ใช้ตัวเลขเมื่อจำนวนนั้นเปลี่ยนสิ่งที่ผู้เล่นจะทำ และใช้จุดเมื่อแค่ "มีอะไรอยู่" ก็พอแล้ว

สีตัวอักษรของมันถูกเลือกให้ตัดกับสีที่สกินทาแบดจ์ไว้ ไม่ว่าจะเป็นสีอะไรก็ตาม — ไม่ได้เดาเอาว่าเป็นสีขาว — และมันมีชื่อเชิงการเข้าถึงติดมาด้วย เพราะจุดสี ๆ จุดเดียวไม่ได้บอกอะไรกับโปรแกรมอ่านหน้าจอเลย

ฟอร์มและรายการ

GoField — แถวที่ผิดพลาดได้

บรรทัดเดียวบนหัวฟอร์มห้าช่องที่บอกว่า "กรุณาตรวจสอบข้อมูลที่กรอก" ไม่ได้บอกว่าช่องไหน เรื่องเดียวเท่านี้แหละที่ทำให้คนเลิกสมัครสมาชิกกลางคัน

var name_field := GoField.make("ชื่อตัวละคร", GoStyle.line_edit("2-12"), "เปลี่ยนภายหลังไม่ได้")
name_field.set_error("ชื่อนี้มีคนใช้แล้ว")   # เซิร์ฟเวอร์ตอบว่าไม่ได้
name_field.clear_error()

คำใบ้จะซ่อนตัวขณะที่ข้อผิดพลาดแสดงอยู่ แถวจึงไม่สูงขึ้นอีกหนึ่งบรรทัดแล้วดันทุกอย่างที่อยู่ข้างใต้ลงไป และข้อความผิดพลาดถูกผนวกเข้าไปในชื่อเชิงการเข้าถึงของคอนโทรล — ขอบสีแดงเพียงอย่างเดียว ไม่ได้บอกอะไรกับคนที่มองไม่เห็นสีแดง

GoStyle.field() คือรุ่นที่อยู่นิ่ง: ป้ายข้อความ คอนโทรล คำใบ้ ไม่มีสถานะผิดพลาด ใช้ตัวโรงงานเมื่อไม่มีอะไรผิดพลาดได้ และใช้โหนดกับทุกอย่างที่เซิร์ฟเวอร์ปฏิเสธได้

GoInputGroup — ช่องกรอกที่เชื่อมติดกับปุ่มของมัน

GoInputGroup.make(GoStyle.line_edit("ข้อความ"), {"suffix": GoStyle.icon_button(&"send", send)})
GoInputGroup.make(GoStyle.line_edit("ชื่อ"), {"prefix_icon": &"search"})

สิ่งที่มันแก้คือมุม รูปทรงมนสองอันที่มาชนกันตรงกลางจะดูเหมือนถูกบีบ และช่องไฟค่าเริ่มต้นทำให้มัน อ่านเป็นของสองชิ้น มุมด้านนอกยังคงมน ส่วนมุมที่ชนกันจะกลายเป็นเหลี่ยม — ในทุกสถานะของปุ่ม ไม่อย่างนั้นกลุ่มนี้จะแยกออกจากกันทันทีที่ถูกกด สกินมุมตัดเฉียงไม่มีฟิลด์มุมอยู่เลย ที่นั่นมันจึงปล่อยไว้ อย่างนั้น แล้วหน้าเหลี่ยมก็มาชนกันได้พอดี

GoCombobox — ตัวเลือกที่ค้นหาได้

ถ้ามีไม่ถึงสิบรายการ GoStyle.select() ดีกว่า: แตะน้อยลงหนึ่งครั้งและเห็นทั้งรายการ แต่พอเกินสามสิบ การกวาดตาหาก็กลายเป็นงาน

var picker := GoCombobox.make(server_names, 0, "เลือกเซิร์ฟเวอร์")
picker.picked.connect(func(i: int) -> void: connect_to(servers[i]))
การค้นหาจับคู่กับข้างในชื่อ ไม่ใช่แค่ตรงต้นชื่อ การจับคู่เฉพาะตัวขึ้นต้นแทบหาอะไรไม่เจอเลย ในรายการภาษาเกาหลีหรือญี่ปุ่น ซึ่งคำที่แยกสองรายการออกจากกันมักไม่ใช่คำแรก

GoCodeInput — โค้ดคูปองและโค้ดของขวัญ

โค้ดถูกวางมาจากข้อความบ่อยกว่าถูกพิมพ์เองมาก ถ้าใช้ช่องกรอกจริงสิบสองช่อง การวางจะพังตั้งแต่ช่องแรก และตัวอักษรจะหายไปกับ IME จึงมีช่องซ่อนอยู่ช่องเดียวที่รับข้อความ ส่วนช่องที่เห็นนั้นเป็นการวาดขึ้นมา ABCD-EFGH-IJKL จะถูกตัดขีดกลางทิ้งระหว่างทางเข้า

var coupon := GoCodeInput.make(12, 4)
coupon.completed.connect(func(code: String) -> void: server.redeem(code))
coupon.set_error("ใช้ไปแล้ว")

GoTable — หัวตารางที่เรียงลำดับได้ แถวที่เลือกได้

var board := GoTable.make(
    [{"text": "อันดับ", "width": 56}, {"text": "ชื่อ"}, {"text": "คะแนน", "numeric": true}], rows)
board.row_selected.connect(func(i: int) -> void: open_profile(rows[i]))
numeric: true ไม่อย่างนั้นอันดับจะกลับหัว ถ้าเทียบกันแบบข้อความ "9124" จะชนะ "91240" — และคะแนน เงินทอง กับความเสียหาย ล้วนมีจำนวนหลักไม่เท่ากันทั้งนั้น

row_selected ให้ดัชนีในอาร์เรย์ต้นฉบับของคุณ ไม่ใช่ตำแหน่งหลังเรียงลำดับ ทิศทางการเรียง แสดงด้วยสัญลักษณ์ ไม่ใช่แค่สี และทั้งแถวกดได้ ไม่มีใครต้องเล็งให้โดนเซลล์เดียว ส่วน GoStyle.table() ยังคงเป็นตารางแบบอ่านอย่างเดียวเหมือนเดิม

GoPagination — แบ่งเป็นหน้า หรือแถว "ดูเพิ่ม"

var pager := GoPagination.make(1, 12, func(page: int) -> void: load_mail(page))
var more := GoPagination.more(load_next)   # รุ่นที่เหมาะกับมือถือ

ตัวเลขหน้าเป็น UI ของเมาส์ บนมือถือแถว "ดูเพิ่ม" อ่านง่ายกว่า หน้าปัจจุบันจะอยู่กึ่งกลางหน้าต่างเสมอ การกดถัดไปจึงไม่สลับตำแหน่งตัวเลขใหม่ทั้งชุด ยอดรวมเป็น 0 หมายถึง "ไม่ทราบ" แล้วมันจะวาดแค่ลูกศร — มันไม่แต่งจำนวนหน้าที่ไม่มีใครบอกมันขึ้นมาเอง

รูปทรงที่เกมใช้จริง

รูปทรงที่คุ้นตาจากเว็บ สร้างขึ้นใหม่รอบสิ่งที่เกมต้องการจากมัน

GoRewardCalendar — การเช็กชื่อรายวัน

ตัวเลือกวันที่แทบไม่มีประโยชน์ในเกมเลย แต่รูปทรงปฏิทินนั้นมี — สำหรับรางวัลเช็กชื่อ ที่สิ่งสำคัญคือ คุณมาถึงวันที่เท่าไรแล้ว ไม่ใช่ว่าวันนี้เป็นวันที่เท่าไร

var attendance := GoRewardCalendar.make(days, claimed_until)
attendance.claimed.connect(func(day: int) -> void: server.claim_day(day))

กดได้เฉพาะช่องของวันนี้ — ปุ่มที่กดแล้วไม่มีอะไรเกิดขึ้นคนอ่านว่ามันพัง ช่องที่รับไปแล้ว ช่องวันนี้ และช่องวันข้างหน้า ต่างกันสามอย่างพร้อมกัน: เครื่องหมายถูก ขอบสีเน้น และการหรี่แสง

GoRadar และ GoDonut — ค่าสถานะที่เห็นได้ในพริบตา

var stats := GoRadar.make({"STR": 0.85, "AGI": 0.5, "INT": 0.3, "VIT": 0.7, "LUK": 0.45})
stats.set_compare(with_new_sword)          # เส้นประที่วาดทับ

var share := GoDonut.make([{"label": "กายภาพ", "value": 620}, {"label": "เวทมนตร์", "value": 340}])
card.add_child(share.legend())

ค่าของเรดาร์เป็น 0–1 เพราะแค่ไหนถึงเรียกว่าเต็มนั้นเป็นการตัดสินใจของเกม — STR 120 ดิบ ๆ วางข้าง INT 45 จะวาดออกมาเป็นคำโกหก ส่วนค่าของโดนัทเป็นค่าดิบและมันหารให้คุณเอง เกินห้าชิ้นเมื่อไรมันจะยุบ ที่เหลือรวมกัน เพราะชิ้นที่หกเป็นเส้นบาง ๆ ที่ไม่มีใครอ่านออก เส้นเปรียบเทียบเป็นเส้นประ ไม่ใช่แค่เปลี่ยนสี และคำอธิบายสัญลักษณ์พิมพ์เปอร์เซ็นต์ออกมาเป็นตัวหนังสือ

GoCarousel — แบนเนอร์และหน้าเลือกตัวละคร

banners.set_pages([promo, event, pack])
banners.autoplay_seconds = 5.0             # ปิดไว้เป็นค่าเริ่มต้น

มันไม่เลื่อนเองถ้าคุณไม่สั่ง และห้าวินาทีคือขั้นต่ำ: แบนเนอร์ที่เปลี่ยนทุกสามวินาทีเอาข้อความไปก่อน ที่จะอ่านจบ การแตะมันจะรีเซ็ตตัวจับเวลา และถ้าเปิด reduce_motion มันจะไม่เลื่อนเองเลย จุดบอกหน้ากดได้เต็มขนาดสัมผัสขั้นต่ำ ถึงแม้จุดที่วาดออกมาจะเล็กก็ตาม

GoKbd — คำใบ้ปุ่มที่ไม่โกหก

hint.add_child(GoKbd.for_action(&"interact"))   # อ่านคีย์ที่ผูกไว้จริง

"กด F" ที่เขียนตายตัวไว้จะยังบอกว่า F ต่อไปหลังผู้เล่นเปลี่ยนปุ่มไปแล้ว — คำโกหกที่พบบ่อยที่สุดใน UI ของ PC for_action() อ่านจาก InputMap ถ้าแอ็กชันนั้นถูกผูกไว้กับเกมแพดแทน มันจะคืนค่าว่าง แล้ววิดเจ็ตจะซ่อนตัว เพราะ "ไม่มี" ก็เป็นคำโกหกของมันเองอีกอย่าง บนแอนดรอยด์และ iOS มันซ่อนตัวเอง

ทับอยู่บนหน้าจอ

GoDrawer — พาเนลด้านข้าง

GoSheet มาจากด้านล่าง ซึ่งถูกต้องสำหรับมือถือที่ถือแนวตั้ง แต่บนแท็บเล็ตหรือเดสก์ท็อป ชีตอันเดียวกันนั้นจะบังเกมไว้ ดรอว์เออร์จึงมาจากด้านข้างแทน

bag.side = GoDrawer.Side.RIGHT
bag.open("กระเป๋า")

ตัวพาเนลยื่นไปจนถึงขอบจอ แต่เนื้อหายังอยู่ในเซฟแอเรีย ไม่อย่างนั้นมุมมนจะตัดแถวแรกหายไป LEFT หมายถึงด้านซ้ายของจอเสมอ แม้ในภาษาอาหรับ เปิด follow_text_direction เมื่อประเด็นคือ "ด้านที่เป็นจุดเริ่ม" แล้วมันจะกลับข้างใน RTL

GoPopover — การ์ดที่ตรึงไว้กับที่

GoPopover.open(slot, item_card(item))

นี่คือการวางแบบตรึงที่ของ GoSurface โดยที่เลเยอร์ ตัวพื้นผิว จุดตรึง และการเก็บกวาด ถูกต่อสายไว้ให้หมดแล้ว — สิบบรรทัดที่ทุกเกมต้องเขียนซ้ำแล้วซ้ำอีก เปิดได้ครั้งละหนึ่งอันเท่านั้น เพราะการ์ดที่ซ้อนกันจะกลบหน้าจอจนไม่มีใครบอกได้ว่าการ์ดไหนเป็นของสล็อตไหน ฉากหลังของมันโปร่งใส: การ์ดที่คุณเปิดขึ้นมาเพื่อเทียบอุปกรณ์ต้องไม่ทำให้อุปกรณ์นั้นมืดลง

มันไม่ใช่ทูลทิปแบบเมาส์ชี้ การสัมผัสไม่มีสถานะ hover อะไรก็ตามที่ต้องใช้เมาส์เท่านั้นถึงจะเปิดให้เห็นได้ จึงมองไม่เห็นเลยสำหรับผู้เล่นครึ่งหนึ่งของคุณ

GoContextMenu — กดค้างและคลิกขวา

GoContextMenu.attach(item_slot, [
    {"text": "ใช้", "action": use},
    {"separator": true},
    {"text": "ทิ้ง", "action": drop, "danger": true},
])

PopupMenu มีอยู่ในเอนจินอยู่แล้ว สิ่งที่ไม่มีคือท่าทาง: กดค้างครึ่งวินาทีเพื่อเปิด และยกเลิก ทันทีที่นิ้วขยับ การเลื่อนรายการจึงไม่มีทางทำให้เมนูเด้งขึ้นมา ส่งเป็น Callable แทนอาร์เรย์เมื่อรายการในเมนูขึ้นอยู่กับสิ่งที่คุณกดลงไป

การกดค้างต้องมีทางเข้าที่สอง ไม่มีใครค้นพบท่าทางที่ซ่อนอยู่ — จับคู่มันกับปุ่ม ⋯ ที่มองเห็นได้ (open_at()) หรือไอคอน หรือโค้ชมาร์กตอนเปิดเกมครั้งแรก

GoConsole — คอนโซลสำหรับนักพัฒนา

console.register("give", "มอบไอเทม: give <id> <count>", give)
console.toggle()

คำสั่งโกงและคำสั่งดีบัก พร้อมพาเลตต์ที่กรองให้ตามที่พิมพ์ จึงไม่มีใครต้องท่องจำมัน และมีประวัติคำสั่ง อยู่บนปุ่มลูกศร มันอยู่เหนือทุกอย่าง คุณจึงดีบักทับไดอะล็อกได้

มันไม่ยอมเปิดในบิลด์รีลีส debug_only เปิดไว้เป็นค่าเริ่มต้น คำสั่งโกงที่ตกไปอยู่ในมือ ผู้เล่นจบระบบเศรษฐกิจของเกมได้ตั้งแต่วันแรก

แถวรายการที่มีคำอธิบาย

GoStyle.list_button(GoIconSet.SWORD, "ดาบคำสาบาน", _inspect, Color.TRANSPARENT,
    "ดาบยาว  /  โจมตี 42", false)

แถวที่มีบรรทัดที่สองจะได้ระยะขอบในด้านบนและล่างเท่ากัน โปร่งกว่าแถวบรรทัดเดียว และคู่ของ หัวเรื่องกับคำอธิบายจะจัดกึ่งกลางแนวตั้งข้างไอคอนเสมอ — รวมถึงตอนที่คอนเทนเนอร์ยืดแถวให้สูงขึ้น และหลังจากคำอธิบายตัดบรรทัดเป็นหลายบรรทัดแล้วด้วย ทั้งแถวยังคงเป็นเป้าหมายการแตะเดียวกัน

แถวที่ไหลต่อกัน

ใส่คอนโทรลลงใน wrap_row ที่ความกว้างตามธรรมชาติของมัน ถ้าปล่อยให้การตัดบรรทัดเปิดอยู่ ความกว้างขั้นต่ำของปุ่มจะลดลงจนแทบเป็นศูนย์ แล้วแถวจะกำหนดขนาดเซลล์ตามนั้น — ป้ายข้อความจะยุบลงเหลือ บรรทัดละตัวอักษร ตัวโรงงานป้องกันเรื่องนี้ให้คุณแล้ว

อย่าปล่อยรายการที่ว่างให้ว่างเปล่า

page.add_child(GoStyle.empty_state(GoIconSet.BOX, "ยังไม่มีอะไรที่นี่"))

คนอ่านหน้าจอเปล่า ๆ ว่ามันพัง
